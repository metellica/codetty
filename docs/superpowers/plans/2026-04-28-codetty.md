# codetty Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a terminal monitoring system where terminal machines connect to a central hub via WebSocket, and the hub displays all sessions in a TUI with grid/fullscreen modes.

**Architecture:** Two binaries — `codetty-cli` (runs on each terminal machine, spawns PTY and connects to hub) and `codetty-hub` (TUI server that displays all connected sessions in a grid, with fullscreen mode for interactive control). Hub uses vt10x for virtual terminal emulation, bubbletea for TUI, and channel-based WebSocket writes for concurrency safety.

**Tech Stack:** Go 1.24, charmbracelet/bubbletea (v1), charmbracelet/lipgloss, gorilla/websocket, creack/pty, hinshun/vt10x, golang.org/x/term

**Spec:** `docs/superpowers/specs/2026-04-28-codetty-design.md`

package protocol

import "encoding/json"

type MsgType byte

const (
	// cli → hub
	MsgRegister MsgType = 'r'
	MsgOutput   MsgType = 'o'

	// hub → cli
	MsgInput  MsgType = 'i'
	MsgResize MsgType = 's'
)

type RegisterPayload struct {
	ID    string `json:"id"`
	Title string `json:"title"`
	Cols  uint16 `json:"cols"`
	Rows  uint16 `json:"rows"`
}

type ResizePayload struct {
	Cols uint16 `json:"cols"`
	Rows uint16 `json:"rows"`
}

func EncodeRegister(p RegisterPayload) ([]byte, error) {
	b, err := json.Marshal(p)
	if err != nil {
		return nil, err
	}
	return append([]byte{byte(MsgRegister)}, b...), nil
}

func DecodeRegister(data []byte) (RegisterPayload, error) {
	var p RegisterPayload
	err := json.Unmarshal(data, &p)
	return p, err
}

func EncodeOutput(data []byte) []byte {
	return append([]byte{byte(MsgOutput)}, data...)
}

func EncodeInput(data []byte) []byte {
	return append([]byte{byte(MsgInput)}, data...)
}

func EncodeResize(cols, rows uint16) ([]byte, error) {
	b, err := json.Marshal(ResizePayload{Cols: cols, Rows: rows})
	if err != nil {
		return nil, err
	}
	return append([]byte{byte(MsgResize)}, b...), nil
}

func DecodeResize(data []byte) (ResizePayload, error) {
	var p ResizePayload
	err := json.Unmarshal(data, &p)
	return p, err
}

// ParseType extracts the message type from a raw WebSocket message.
// Returns the type and the payload (everything after the type byte).
func ParseType(msg []byte) (MsgType, []byte, bool) {
	if len(msg) < 1 {
		return 0, nil, false
	}
	return MsgType(msg[0]), msg[1:], true
}

package hub

import (
	"log"

	"github.com/gorilla/websocket"
)

const writeChanSize = 256

// ConnWriter serializes WebSocket write operations through a channel.
// gorilla/websocket does not support concurrent writes; all writes for
// a connection must go through a single ConnWriter.
type ConnWriter struct {
	conn    *websocket.Conn
	writeCh chan []byte
	done    chan struct{}
}

func NewConnWriter(conn *websocket.Conn) *ConnWriter {
	w := &ConnWriter{
		conn:    conn,
		writeCh: make(chan []byte, writeChanSize),
		done:    make(chan struct{}),
	}
	go w.writeLoop()
	return w
}

func (w *ConnWriter) writeLoop() {
	defer close(w.done)
	for msg := range w.writeCh {
		if err := w.conn.WriteMessage(websocket.BinaryMessage, msg); err != nil {
			log.Printf("ws write error: %v", err)
			return
		}
	}
}

// Send enqueues a message for writing. Non-blocking; drops the message
// if the channel is full (slow client protection).
func (w *ConnWriter) Send(data []byte) {
	select {
	case w.writeCh <- data:
	default:
		log.Printf("write channel full, dropping message (%d bytes)", len(data))
	}
}

// Close stops the write loop and waits for it to finish.
func (w *ConnWriter) Close() {
	close(w.writeCh)
	<-w.done
}
```

package hub

import (
	"codetty/protocol"

	"github.com/hinshun/vt10x"
)

// ClientSession represents one connected terminal client.
type ClientSession struct {
	ID    string
	Title string
	Cols  uint16
	Rows  uint16
	term  vt10x.Terminal
	writer *ConnWriter
}

// NewClientSession creates a session with a vt10x virtual terminal.
// Pass nil for writer if no WebSocket connection is available (testing).
func NewClientSession(id, title string, cols, rows uint16, writer *ConnWriter) *ClientSession {
	term := vt10x.New(vt10x.WithSize(int(cols), int(rows)))
	return &ClientSession{
		ID:     id,
		Title:  title,
		Cols:   cols,
		Rows:   rows,
		term:   term,
		writer: writer,
	}
}

// WriteOutput feeds raw PTY output bytes into the vt10x terminal.
func (s *ClientSession) WriteOutput(data []byte) {
	s.term.Write(data)
}

// Screen returns the current virtual terminal screen content as a string.
func (s *ClientSession) Screen() string {
	s.term.Lock()
	defer s.term.Unlock()
	return s.term.String()
}

// SendInput sends keyboard input to the remote client.
func (s *ClientSession) SendInput(data []byte) {
	if s.writer != nil {
		s.writer.Send(protocol.EncodeInput(data))
	}
}

// SendResize sends a resize command to the remote client and updates local state.
func (s *ClientSession) SendResize(cols, rows uint16) {
	s.Cols = cols
	s.Rows = rows
	s.term.Resize(int(cols), int(rows))
	if s.writer != nil {
		msg, err := protocol.EncodeResize(cols, rows)
		if err == nil {
			s.writer.Send(msg)
		}
	}
}

// Close releases resources. Safe to call multiple times.
func (s *ClientSession) Close() {
	if s.writer != nil {
		s.writer.Close()
		s.writer = nil
	}
}
```

package hub

import (
	"log"
	"os"
	"sync"

	tea "github.com/charmbracelet/bubbletea"
	"github.com/gorilla/websocket"

	"codetty/protocol"
)

// Tea messages sent from Hub to TUI
type ClientConnectedMsg struct{ Session *ClientSession }
type ClientDisconnectedMsg struct{ ID string }
type ScreenUpdateMsg struct {
	ID   string
	Data []byte
}

// Hub manages all connected client sessions.
type Hub struct {
	mu       sync.RWMutex
	sessions map[string]*ClientSession
	order    []string // ordered list of session IDs
	program  *tea.Program

	fsMu         sync.RWMutex
	fullscreenID string
}

func New() *Hub {
	return &Hub{
		sessions: make(map[string]*ClientSession),
	}
}

func (h *Hub) SetProgram(p *tea.Program) {
	h.program = p
}

// SetFullscreen sets which session is currently displayed in fullscreen mode.
// Pass empty string to exit fullscreen.
func (h *Hub) SetFullscreen(id string) {
	h.fsMu.Lock()
	h.fullscreenID = id
	h.fsMu.Unlock()
}

func (h *Hub) GetFullscreenID() string {
	h.fsMu.RLock()
	defer h.fsMu.RUnlock()
	return h.fullscreenID
}

// HandleConn handles a new WebSocket connection from a client.
// Blocks until the connection is closed.
func (h *Hub) HandleConn(conn *websocket.Conn) {
	var sess *ClientSession

	defer func() {
		if sess != nil {
			h.removeSession(sess.ID)
		}
		conn.Close()
	}()

	for {
		_, msg, err := conn.ReadMessage()
		if err != nil {
			return
		}

		msgType, payload, ok := protocol.ParseType(msg)
		if !ok {
			return
		}

		switch msgType {
		case protocol.MsgRegister:
			p, err := protocol.DecodeRegister(payload)
			if err != nil {
				log.Printf("bad register payload: %v", err)
				return
			}
			writer := NewConnWriter(conn)
			sess = NewClientSession(p.ID, p.Title, p.Cols, p.Rows, writer)
			h.addSession(sess)

		case protocol.MsgOutput:
			if sess == nil {
				return
			}
			sess.WriteOutput(payload)

			// In fullscreen mode, also write raw output directly to stdout
			if h.GetFullscreenID() == sess.ID {
				os.Stdout.Write(payload)
			}

			if h.program != nil {
				h.program.Send(ScreenUpdateMsg{ID: sess.ID, Data: payload})
			}
		}
	}
}

func (h *Hub) addSession(s *ClientSession) {
	h.mu.Lock()
	// If a session with the same ID exists, close the old one
	if old, exists := h.sessions[s.ID]; exists {
		old.Close()
		// Replace in order slice
		for i, id := range h.order {
			if id == s.ID {
				h.order = append(h.order[:i], h.order[i+1:]...)
				break
			}
		}
	}
	h.sessions[s.ID] = s
	h.order = append(h.order, s.ID)
	h.mu.Unlock()

	if h.program != nil {
		h.program.Send(ClientConnectedMsg{Session: s})
	}
}

func (h *Hub) removeSession(id string) {
	h.mu.Lock()
	if sess, exists := h.sessions[id]; exists {
		sess.Close()
		delete(h.sessions, id)
		for i, oid := range h.order {
			if oid == id {
				h.order = append(h.order[:i], h.order[i+1:]...)
				break
			}
		}
	}
	h.mu.Unlock()

	if h.program != nil {
		h.program.Send(ClientDisconnectedMsg{ID: id})
	}
}

// Sessions returns an ordered copy of all active sessions.
func (h *Hub) Sessions() []*ClientSession {
	h.mu.RLock()
	defer h.mu.RUnlock()
	out := make([]*ClientSession, 0, len(h.order))
	for _, id := range h.order {
		if s, ok := h.sessions[id]; ok {
			out = append(out, s)
		}
	}
	return out
}

// Session returns a session by ID, or nil if not found.
func (h *Hub) Session(id string) *ClientSession {
	h.mu.RLock()
	defer h.mu.RUnlock()
	return h.sessions[id]
}

		m.width = msg.Width
		m.height = msg.Height

	case ClientConnectedMsg:
		m.sessions = m.hub.Sessions()

	case ClientDisconnectedMsg:
		m.sessions = m.hub.Sessions()
		if m.selected >= len(m.sessions) {

		}

	case tea.KeyMsg:
		if m.fullscreen {
			// This shouldn't happen — fullscreen uses tea.Exec which
			// takes over input. But handle gracefully:
			return m, nil
		}
		switch msg.String() {
		case "q", "ctrl+c":
			return m, tea.Quit
		case "tab", "l", "right":
			if len(m.sessions) > 0 {
				m.selected = (m.selected + 1) % len(m.sessions)
			}
		case "shift+tab", "h", "left":
			if len(m.sessions) > 0 {
				m.selected = (m.selected - 1 + len(m.sessions)) % len(m.sessions)
			}
		case "j", "down":
			if len(m.sessions) > 0 {
				cols := m.gridCols()
				next := m.selected + cols
				if next < len(m.sessions) {
					m.selected = next
				}
			}
		case "k", "up":
			if len(m.sessions) > 0 {
				cols := m.gridCols()
				next := m.selected - cols
				if next >= 0 {
					m.selected = next
				}
			}
		case "enter", "f":
			if len(m.sessions) > 0 {
				m.fullscreen = true
				sess := m.sessions[m.selected]
				// Resize remote PTY to match our terminal
				sess.SendResize(uint16(m.width), uint16(m.height))
				handler := &fullscreenHandler{
					session: sess,
					hub:     m.hub,
				}
				return m, tea.Exec(handler, func(err error) tea.Msg {
					return exitFullscreenMsg{}
				})
			}
		}
	}
	return m, nil

func (m *Model) gridCols() int {

	if n <= 1 {
		return 1
	}
	if n <= 4 {
		return 2
	}
	return 3

func (m *Model) panelSize() (int, int) {
	n := len(m.sessions)
	if n == 0 {
		return m.width - 4, m.height - 4
	}
	cols := m.gridCols()
	rows := (n + cols - 1) / cols
	panelW := (m.width / cols) - 2
	panelH := ((m.height - 2) / rows) - 2
	if panelW < 10 {
		panelW = 10
	}
	if panelH < 3 {
		panelH = 3
	}
	return panelW, panelH
}

func (m *Model) gridView() string {
	n := len(m.sessions)
	cols := m.gridCols()
	rows := (n + cols - 1) / cols
	panelW, panelH := m.panelSize()

	var rowViews []string
	for r := 0; r < rows; r++ {
		var colViews []string
		for c := 0; c < cols; c++ {
			idx := r*cols + c
			if idx >= n {
				colViews = append(colViews, strings.Repeat(" ", panelW+4))
				continue
			}
			sess := m.sessions[idx]
			content := renderScreen(sess.Screen(), panelW, panelH-1)
			title := fmt.Sprintf(" [%d] %s ", idx+1, sess.Title)

			style := panelStyle
			if idx == m.selected {
				style = focusedPanelStyle
				title = fmt.Sprintf(" ▶ [%d] %s ", idx+1, sess.Title)
			}

			panel := style.Width(panelW).Height(panelH).Render(
				titleStyle.Render(title) + "\n" + content,
			)
			colViews = append(colViews, panel)
		}
		rowViews = append(rowViews, lipgloss.JoinHorizontal(lipgloss.Top, colViews...))
	}

	grid := lipgloss.JoinVertical(lipgloss.Left, rowViews...)
	help := statusBarStyle.Width(m.width).Render(
		fmt.Sprintf("Tab/←→/↑↓: navigate  Enter/f: fullscreen  q: quit  clients: %d", n),
	)
	return lipgloss.JoinVertical(lipgloss.Left, grid, help)
}

func renderScreen(screen string, w, h int) string {
	lines := strings.Split(screen, "\n")
	// Take last h lines
	if len(lines) > h {
		lines = lines[len(lines)-h:]
	}
	var sb strings.Builder
	for i, l := range lines {
		if i > 0 {
			sb.WriteRune('\n')
		}
		// Truncate to panel width
		runes := []rune(l)
		if len(runes) > w {
			runes = runes[:w]
		}
		sb.WriteString(string(runes))
	}
	return sb.String()
}

package hub

import (
	"io"
	"os"
	"time"
)

// fullscreenHandler implements tea.ExecCommand. When bubbletea runs it
// via tea.Exec, the terminal is released and we get direct access to
// stdin/stdout for raw terminal proxying.
type fullscreenHandler struct {
	session *ClientSession
	hub     *Hub
	stdin   io.Reader
	stdout  io.Writer
}

func (f *fullscreenHandler) SetStdin(r io.Reader)  { f.stdin = r }
func (f *fullscreenHandler) SetStdout(w io.Writer) { f.stdout = w }
func (f *fullscreenHandler) SetStderr(w io.Writer) {}

// Run proxies raw input from the hub's terminal to the remote session.
// Output is handled by Hub.HandleConn writing to os.Stdout when
// fullscreen is active. Blocks until the user presses ESC-ESC or Ctrl+A.
func (f *fullscreenHandler) Run() error {
	// Tell the hub we're in fullscreen so HandleConn writes to stdout
	f.hub.SetFullscreen(f.session.ID)
	defer f.hub.SetFullscreen("")

	// Use os.Stdin directly for raw reading (bubbletea has released it)
	stdin := os.Stdin

	buf := make([]byte, 4096)
	var lastEsc time.Time

	for {
		n, err := stdin.Read(buf)
		if err != nil {
			return err
		}
		data := buf[:n]

		// Detect exit sequences
		if len(data) > 0 {
			// Ctrl+A (byte 0x01) exits immediately
			if data[0] == 0x01 {
				return nil
			}
			// Standalone ESC (0x1b as single byte) — double ESC within 300ms exits
			if data[0] == 0x1b && len(data) == 1 {
				if time.Since(lastEsc) < 300*time.Millisecond {
					return nil
				}
				lastEsc = time.Now()
			}
		}

		// Forward all input to the remote session's PTY
		f.session.SendInput(data)
	}
}

package main

import (
	"flag"
	"fmt"
	"log"
	"os"
	"os/exec"
	"os/signal"
	"syscall"
	"time"

	"codetty/protocol"

	"github.com/creack/pty"
	"github.com/gorilla/websocket"
	xterm "golang.org/x/term"
)

func main() {
	serverURL := flag.String("server", "ws://localhost:8080/ws", "hub WebSocket URL")
	clientID := flag.String("id", mustHostname(), "client identifier")
	flag.Parse()

	args := flag.Args()
	if len(args) == 0 {
		fmt.Fprintf(os.Stderr, "Usage: codetty-cli --server ws://hub:8080/ws --id NAME -- <command> [args...]\n")
		os.Exit(1)
	}

	// Determine terminal size
	cols, rows := uint16(220), uint16(50)
	if xterm.IsTerminal(int(os.Stdin.Fd())) {
		w, h, err := xterm.GetSize(int(os.Stdin.Fd()))
		if err == nil {
			cols, rows = uint16(w), uint16(h)
		}
	}

	// Spawn the command in a PTY
	cmd := exec.Command(args[0], args[1:]...)
	ptmx, err := pty.StartWithSize(cmd, &pty.Winsize{Cols: cols, Rows: rows})
	if err != nil {
		log.Fatalf("pty.Start: %v", err)
	}
	defer ptmx.Close()

	title := fmt.Sprintf("%s@%s", args[0], *clientID)

	// Signal handling
	sigCh := make(chan os.Signal, 1)
	signal.Notify(sigCh, syscall.SIGTERM, syscall.SIGINT)

	// Process exit channel
	doneCh := make(chan error, 1)
	go func() { doneCh <- cmd.Wait() }()

	// Run connection loop (reconnects on failure)
	go connectionLoop(*serverURL, *clientID, title, cols, rows, ptmx, doneCh)

	// Wait for process exit or signal
	select {
	case <-sigCh:
		log.Println("received signal, shutting down")
	case err := <-doneCh:
		if err != nil {
			log.Printf("command exited: %v", err)
		} else {
			log.Println("command exited normally")
		}
	}
}

func connectionLoop(serverURL, clientID, title string, cols, rows uint16, ptmx *os.File, done <-chan error) {
	delay := time.Second
	maxDelay := 30 * time.Second

	for {
		err := runConnection(serverURL, clientID, title, cols, rows, ptmx, done)
		if err == nil {
			// Clean disconnect or process exit
			return
		}

		log.Printf("connection lost: %v, reconnecting in %v", err, delay)

		select {
		case <-done:
			return
		case <-time.After(delay):
		}

		// Exponential backoff
		delay *= 2
		if delay > maxDelay {
			delay = maxDelay
		}
	}
}

func runConnection(serverURL, clientID, title string, cols, rows uint16, ptmx *os.File, done <-chan error) error {
	conn, _, err := websocket.DefaultDialer.Dial(serverURL, nil)
	if err != nil {
		return fmt.Errorf("dial: %w", err)
	}
	defer conn.Close()

	// Register with hub
	regMsg, err := protocol.EncodeRegister(protocol.RegisterPayload{
		ID:    clientID,
		Title: title,
		Cols:  cols,
		Rows:  rows,
	})
	if err != nil {
		return fmt.Errorf("encode register: %w", err)
	}
	if err := conn.WriteMessage(websocket.BinaryMessage, regMsg); err != nil {
		return fmt.Errorf("send register: %w", err)
	}

	// Reset backoff on successful connection
	log.Printf("connected to %s as %s", serverURL, clientID)

	errCh := make(chan error, 2)

	// PTY output → hub
	go func() {
		buf := make([]byte, 4096)
		for {
			n, err := ptmx.Read(buf)
			if n > 0 {
				msg := protocol.EncodeOutput(buf[:n])
				if werr := conn.WriteMessage(websocket.BinaryMessage, msg); werr != nil {
					errCh <- fmt.Errorf("write output: %w", werr)
					return
				}
			}
			if err != nil {
				errCh <- nil // PTY closed = clean exit
				return
			}
		}
	}()

	// Hub → PTY input / resize
	go func() {
		for {
			_, msg, err := conn.ReadMessage()
			if err != nil {
				errCh <- fmt.Errorf("read: %w", err)
				return
			}
			msgType, payload, ok := protocol.ParseType(msg)
			if !ok {
				continue
			}
			switch msgType {
			case protocol.MsgInput:
				ptmx.Write(payload)
			case protocol.MsgResize:
				p, err := protocol.DecodeResize(payload)
				if err == nil {
					pty.Setsize(ptmx, &pty.Winsize{Cols: p.Cols, Rows: p.Rows})
				}
			}
		}
	}()

	// Wait for error from either goroutine, or process exit
	select {
	case err := <-errCh:
		return err
	case <-done:
		conn.WriteMessage(websocket.CloseMessage,
			websocket.FormatCloseMessage(1000, "process exited"))
		return nil
	}
}

func mustHostname() string {
	h, _ := os.Hostname()
	if h == "" {
		return "unknown"
	}
	return h
}

	flag.Parse()

			log.Printf("upgrade error: %v", err)
			return
		}
		go h.HandleConn(conn)

	go func() {
		log.Printf("codetty-hub listening on %s", *addr)
		if err := http.ListenAndServe(*addr, nil); err != nil {
			log.Fatalf("http server: %v", err)
		}
	}()

	// Run TUI (blocks until quit)
	if _, err := p.Run(); err != nil {
